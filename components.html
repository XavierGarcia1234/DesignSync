<!-- HTML -->
<button class="theme-toggle-btn" id="theme-toggle" aria-label="Toggle theme">
    <svg class="sun-icon">...</svg>
    <svg class="moon-icon">...</svg>
</button>

/* CSS */
.theme-toggle-btn {
    background: none; border: none; cursor: pointer;
    color: var(--ink-muted);
    display: flex; align-items: center; justify-content: center;
    position: relative; width: 32px; height: var(--nav-height);
}
.theme-toggle-btn:hover { color: var(--ink); }
.theme-toggle-btn svg {
    position: absolute;
    transition: opacity 0.4s ease, transform 0.4s ease;
}
.sun-icon { opacity: var(--opacity-0); transform: rotate(-90deg) scale(var(--scale-50)); }
.moon-icon { opacity: var(--opacity-100); transform: rotate(0) scale(var(--scale-100)); }
[data-theme="light"] .sun-icon { opacity: var(--opacity-100); transform: rotate(0) scale(var(--scale-100)); }
[data-theme="light"] .moon-icon { opacity: var(--opacity-0); transform: rotate(90deg) scale(var(--scale-50)); }

<!-- Javascript -->
<script>
const themeToggle = document.getElementById('theme-toggle');
themeToggle.addEventListener('click', () => {
    document.documentElement.classList.add('theme-transition');
    let theme = document.documentElement.getAttribute('data-theme');
    if (theme === 'light') {
        document.documentElement.removeAttribute('data-theme');
        localStorage.setItem('theme', 'dark');
    } else {
        document.documentElement.setAttribute('data-theme', 'light');
        localStorage.setItem('theme', 'light');
    }
    setTimeout(() => { document.documentElement.classList.remove('theme-transition'); }, 400);
});
</script>

    if (e.target.checked) {
        document.body.classList.add('a11y-large-text');
        localStorage.setItem('a11y-large-text', 'true');
    } else {
        document.body.classList.remove('a11y-large-text');
        localStorage.setItem('a11y-large-text', 'false');
    }
});
